โครงสร้างฐานข้อมูล MySQL (Database Schema)

โค้ด SQL DDL สำหรับสร้างตารางฐานข้อมูลในระบบหลังบ้าน

-- 1. ตารางหมวดหมู่อาหาร (categories)
CREATE TABLE categories (
    category_id INT AUTO_INCREMENT PRIMARY KEY,
    category_name VARCHAR(100) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 2. ตารางเมนูอาหาร (menus)
CREATE TABLE menus (
    menu_id INT AUTO_INCREMENT PRIMARY KEY,
    category_id INT,
    title VARCHAR(150) NOT NULL,
    description TEXT,
    price DECIMAL(10, 2) NOT NULL,
    spicy_level INT DEFAULT 0,
    image_url VARCHAR(255),
    is_available TINYINT(1) DEFAULT 1,
    is_recommend TINYINT(1) DEFAULT 0,
    FOREIGN KEY (category_id) REFERENCES categories(category_id) ON DELETE SET NULL
);

-- 3. ตารางผู้ใช้งาน / ลูกค้า (users)
CREATE TABLE users (
    user_id INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    full_name VARCHAR(100),
    phone VARCHAR(20),
    role ENUM('customer', 'admin') DEFAULT 'customer',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 4. ตารางคำสั่งซื้อ (orders)
CREATE TABLE orders (
    order_id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NULL,
    customer_name VARCHAR(100) NOT NULL,
    customer_phone VARCHAR(20) NOT NULL,
    order_type ENUM('dine-in', 'takeaway', 'delivery') NOT NULL,
    table_number VARCHAR(20) NULL,
    delivery_address TEXT NULL,
    special_notes TEXT NULL,
    total_amount DECIMAL(10, 2) NOT NULL,
    order_status ENUM('pending', 'cooking', 'completed', 'cancelled') DEFAULT 'pending',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 5. ตารางรายละเอียดรายการในคำสั่งซื้อ (order_items)
CREATE TABLE order_items (
    item_id INT AUTO_INCREMENT PRIMARY KEY,
    order_id INT NOT NULL,
    menu_id INT NOT NULL,
    quantity INT NOT NULL,
    unit_price DECIMAL(10, 2) NOT NULL,
    FOREIGN KEY (order_id) REFERENCES orders(order_id) ON DELETE CASCADE,
    FOREIGN KEY (menu_id) REFERENCES menus(menu_id)
);

ตัวอย่างโค้ดระบบหลังบ้าน PHP (api/orders.php) 

<?php
// db.php - การเชื่อมต่อฐานข้อมูล MySQL
header('Content-Type: application/json; charset=utf-8');
$host = "localhost";
$user = "root";
$pass = "";
$db   = "krua_thai_imsook";

$conn = new mysqli($host, $user, $pass, $db);
if ($conn->connect_error) {
    die(json_encode(["success" => false, "message" => "Connection failed: " . $conn->connect_error]));
}

// ตัวอย่าง API รับคำสั่งซื้อใหม่จากหน้าเว็บ
if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $data = json_decode(file_get_contents("php://input"), true);
    
    $name = $conn->real_escape_string($data['customer_name']);
    $phone = $conn->real_escape_string($data['customer_phone']);
    $order_type = $conn->real_escape_string($data['order_type']);
    $total = $data['total_amount'];
    
    $sql = "INSERT INTO orders (customer_name, customer_phone, order_type, total_amount, order_status) 
            VALUES ('$name', '$phone', '$order_type', '$total', 'pending')";
            
    if ($conn->query($sql) === TRUE) {
        $order_id = $conn->insert_id;
        echo json_encode(["success" => true, "order_id" => $order_id, "message" => "บันทึกคำสั่งซื้อเรียบร้อย"]);
    } else {
        echo json_encode(["success" => false, "message" => $conn->error]);
    }
}
?>
